
\section{Move Semantics}

\begin{frame}[fragile]{Copy Construction}
  \begin{sqt:definition}
    A copy constructor is a constructor that takes an existing object of the same class as its argument.
  \end{sqt:definition}
  Defining a copy constructor allows code like this to compile:
  \begin{minted}{cpp}
    /* 1.01 */
    auto a = Gadget{};
    auto& aLref = a;
    auto&& aRref = std::move(a);

    {
        // all these are copy constructions
        auto c = Gadget{a};
        auto d = Gadget{aLref};
        auto e = Gadget{aRref};
        // this is copy assignment, not copy construction
        e = a;
    }
  \end{minted}
\end{frame}

\begin{frame}[fragile]{Copy Construction Convention}
  \begin{sqt:result}
    In a well-designed class, copy construction should create a "fully independent duplicate", such that
    both objects can be used and destroyed independently.
  \end{sqt:result}

  The compiler auto-generates a copy-ctor that recursively copies each member.
  \begin{minted}{cpp}
    struct BadIntBox {
      int *mData;
      BadIntBox(int x) : mData{new int(x)} {}
      ~BadIntBox() { delete mData; }
      // auto-generated copy constructor
      BadIntBox(const BadIntBox& other) : mData{other.mData} {}
    };
  \end{minted}
  Usually, automatically generated copy constructors are NOT well-designed.\\
  In this example, the resultant copy cannot be used nor destroyed independently.
\end{frame}

\begin{frame}[fragile]{Double Frees}
  Double frees occur when you write buggy classes that don't handle copying properly.

  Consider the below example using the \code{BadIntBox} we developed earlier
  \begin{minted}{cpp}
    int main() {
        auto original = BadIntBox{42};
        auto copy = BadIntBox{original}; // copy.mData = original.mData (both point to same heap object!)
    
        // ~original; ~copy; -> Double Free!
    }
  \end{minted}
  When the destructors run, both objects think that they are responsible for destroying
  the heap-allocated integer and freeing it. This results in Undefined Behavior!
  
  To fix this, think carefully when implementing your copy constructors.\\
  Make your class non-copyable if applicable.
\end{frame}

\begin{frame}[fragile]{Copy Constructor Impl}
  To perform a proper \emph{deep copy} on our \code{IntBox}, we want
  \begin{itemize}
    \item a new region of memory to be allocated on the heap
    \item the \code{other.mData} integer is copy constructed into this new region
  \end{itemize}

  \begin{minted}{cpp}
    struct IntBox {
      int *mData;
      IntBox(int x) : mData{new int(x)} {}
      ~IntBox() { delete mData; }
      IntBox(const IntBox& other) : mData{new int(*(other.mData))} {}
    };
  \end{minted}
\end{frame}

\begin{frame}[fragile]{Value Categories}
  Every \emph{expression} has a \emph{value category} (lvalue or rvalue).
  \begin{minted}{cpp}
    struct Lorem { struct Ipsum { int x; } y; int z; } l;
    auto* ptr = &l.z;
    int arr[3] = {1, 2, 3};
  \end{minted}

  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{lvalues} — has identity (an address)
    \begin{minted}{cpp}
l             // named variable
l.z           // member access
l.y.x         // nested member
*ptr          // dereference
arr[0]        // subscript
    \end{minted}
    Can appear on the \emph{left} side of assignment.\\
    Can take the address with \code{\&}.

    \column{0.48\textwidth}
    \textbf{rvalues} — no persistent identity
    \begin{minted}{cpp}
42            // literal
l.z + 1       // arithmetic result
IntBox(3)     // temporary object
std::move(l)  // cast to rvalue
int{}         // value-initialized tmp
    \end{minted}
    Cannot appear on the left side of assignment.\\
    Cannot take the address with \code{\&}.
  \end{columns}
\end{frame}

\begin{frame}[fragile]{rvalue references}
  \begin{sqt:definition}
    An \emph{rvalue reference} is a reference that can bind to an \emph{rvalue}.
  \end{sqt:definition}
  \begin{sqt:result}
    When a ref is bound to a tmp, lifetime extended until the ref goes out of scope.\\
    i.e. the compiler postpones the dtor to the end of the scope.
  \end{sqt:result}

  \begin{minted}{cpp}
    /* 2.01 */
    {
        // tell compiler: we want to extend the tmp's lifetime until end of scope
        IntBox&& rref = IntBox(3);
        // play around with the temporary via the rvalue ref
        std::cout << rref.get() << std::endl;
        rref.get() *= 10;
        std::cout << rref.get() << std::endl;
    }
  \end{minted}
\end{frame}

\begin{frame}[fragile]{Arguments by rvalue ref}
  We already know that a function can take arguments by (lvalue) reference.\\
  It is also possible for a function to take arguments by rvalue reference.

  \begin{minted}{cpp}
/* 2.01 */
void foo(IntBox& obj) { obj.mData += 42; }
void foo(IntBox&& obj) { obj.mData = nullptr; }

int main() {
    // If you pass in an rvalue (e.g. the tmp result of the ctor function IntBox(3)),
    // the rvalue ref extends the lifetime of the tmp until the function returns.
    foo(IntBox(3));
}
  \end{minted}
\end{frame}

\begin{frame}[fragile]{Reference Binding}
  \begin{columns}[T]
    \column{0.64\textwidth}
    \begin{minted}{cpp}
/* 2.01 */
{
    IntBox{3};                     // dies here
    auto&& rref = IntBox{3};       // extended
    const auto& lref = IntBox{3};  // extended
    auto x = IntBox{3};            // no tmp
}   // x, lref, rref destroyed here
{
    auto& data = IntBox{3}.get();  // UB!
    std::cout << data << std::endl;
}
    \end{minted}

    \column{0.32\textwidth}
    \textbf{Output}
    \begin{minted}{text}
constructor
destructor
constructor
constructor
constructor
destructor
destructor
destructor
constructor
destructor
894751456
    \end{minted}
  \end{columns}

  \begin{sqt:result}
    \code{auto\&\&} and \code{const auto\&} extend a tmp's lifetime, \code{auto\&} cannot bind to tmps.\\
    Only \emph{direct} binding extends: \code{get()} returns a ref to an object that dies immediately.
  \end{sqt:result}
\end{frame}

\begin{frame}[fragile]{Overload resolution}
  Given that an argument has two overloads, one taking lvalues and one taking rvalues,
  how does the compiler determine which overload to dispatch for each call-site?

  \begin{minted}{cpp}
    {
        auto x = IntBox{3};
        IntBox&& rref = std::move(x);

        foo(x); // calls lvalue overload
        foo(rref); // also calls lvalue overload (weird)
        foo(IntBox(3)); // calls rvalue overload
        foo(std::move(x)); // calls rvalue overload
        foo(std::move(rref)); // calls rvalue overload
    }
  \end{minted}

  \begin{sqt:result}
    The compiler picks the overload depending on the \emph{value category} of the argument.
  \end{sqt:result}
\end{frame}

\begin{frame}[fragile]{Move Construction}
  \begin{sqt:definition}
    A move constructor is a constructor that takes an rvalue reference
    to an object of the same type.
  \end{sqt:definition}

  \begin{minted}{cpp}
    struct IntBox {
        // ctor 
        IntBox(int x) : mData{new int(x)} {}
        // copy ctor
        IntBox(const IntBox& other) : mData{new int(*(other.mData))} {}
        // move ctor
        IntBox(IntBox&& other) : mData{std::exchange(other.mData, nullptr)} {}
    };

    int main() {
        auto original = IntBox{42};
        auto copied = IntBox{original};
        auto moved = IntBox{static_cast<IntBox&&>(original)};
    }
  \end{minted}
\end{frame}

\begin{frame}[fragile]{Move Construction Convention}
  \begin{sqt:result}
    In a well-designed class, move ctor should transfer ownership of resources from
    \code{other} to \code{this}, such that upon completion, \code{this} represents the same abstract entity
    that \code{other} used to, whilst \code{other} should end up in a well-formed null-like state,
    such that its dtor may be called without issue.
  \end{sqt:result}

  \begin{minted}{cpp}
    struct IntBox {
        // poorly designed move constructor, absolute nonsense
        IntBox(IntBox&& other) : mData{(int*) 12345} {}

        // poorly designed move constructor, shares ownership, double-free and UB
        IntBox(IntBox&& other) : mData{other.mData} {}

        // poorly designed move constructor, transfers ownership
        // but `other` is no longer well-formed. UB upon destruction.
        IntBox(IntBox&& other) : mData{std::exchange(other.mData, (int*) 12345)} {}
    };
  \end{minted}
\end{frame}
