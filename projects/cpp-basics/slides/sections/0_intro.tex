\section{Introduction}

\begin{frame}[fragile]{Before we start}
  Prerequisites
  \begin{itemize}
    \item basic OOP in C++ (methods, initialization, constructors)
    \item basic assembly concepts (labels, stack frames)
  \end{itemize}

  Workshop topics
  \begin{itemize}
    \item Objects, RAII
    \item Move semantics, temporaries, RVO
    \item Templates, universal references
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Conventions and Notation}
  \begin{itemize}
    \item will use x86-64, System V ABI, -O0
    \item I will sometimes refer to methods, constructors and destructors by their full demangled function name e.g.
      \mintinline{cpp}{ExampleClassName::exampleMethodName(arg1, arg2)}.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Visualizations}
  \begin{itemize}
    \item We will sometimes visualize the state of a program at a snapshot in time.
    \item The stack will be drawn as growing downwards.
  \end{itemize}
\end{frame}
